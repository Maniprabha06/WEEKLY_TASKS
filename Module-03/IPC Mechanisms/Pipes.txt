Pipes:
    - simple Memos
    - one way communication
    - like passing a memo from one office to another.
    - The sender writes the memo and the receiver reads it.
    - pipe() system call is used to create a communication channel.
    - use fork() system call to create a child process.
    - The parent sends the message 
    - The child receives the message.

    command :
        - ls | grep "bash"
        - ls (stdout) -> (pipe) -> grep (stdin).

   - nano pipe_program.c

   pipe_program.c
   #include <stdio.h>
   #include <stdlib.h>
   #include <unistd.h>
   #include <string.h>

   int main() {
        int fd[2];  //fd[0]= read end fd[1]=write end
        pid_t pid;
        char buffer[100];
        char msg[]= "Hello from parent process!";

        if(pipe(fd)==-1){
            perror("pipe");
            exit(1);
        }
        pid = fork();
        if(pid < 0){
            perror("fork");
            exit(1);   
        }

        // pid = 12345 or else pid=0
        if(pid > 0){
            //parent process
            close(fd[0]);
            write(fd[1],msg,strlen(msg)+1);
            close(fd[1]);
        }
        else{
            //child process
            close(fd[1]);
            read(fd[0],buffer,sizeof(buffer));
            printf("Child received: %s\n",buffer);
            close(fd[0]);
        }
   }
